<script>
function fmtEur(n) {
    return Number(n || 0).toLocaleString('it-IT', { style: 'currency', currency: 'EUR' });
}
function setMsg(id, text) {
    const el = document.getElementById(id);
    if (!el) return;
    el.style.display = 'block';
    el.textContent = text;
}
function initTabs() {
    const btns = document.querySelectorAll('.tab-btn');
    const panels = document.querySelectorAll('.tab-panel');
    btns.forEach(btn => {
        btn.addEventListener('click', () => {
            const target = btn.getAttribute('data-tab');
            btns.forEach(b => b.classList.remove('active'));
            panels.forEach(p => p.classList.remove('active'));
            btn.classList.add('active');
            const panel = document.getElementById(target);
            if (panel) panel.classList.add('active');
        });
    });
}
initTabs();
const KEY = 'ecs_full_standard_v1';
const def = {
    az: { rag: '', piva: '', sede: '' },
    cantieri: [{ cod: 'CNT-001', nome: 'Cantiere Esempio', comm: 'Committente Esempio', imp: 35000, sal: '25%', stato: 'Attivo' }],
    anag: [{ tipo: 'Cliente', nome: 'Committente Esempio', piva: '-', cont: 'Referente cantiere' }]
};
let st = JSON.parse(localStorage.getItem(KEY) || 'null') || structuredClone(def);
function persist() { localStorage.setItem(KEY, JSON.stringify(st)); render(); }

function render() {
    document.getElementById('azRag').value = st.az.rag || '';
    document.getElementById('azPiva').value = st.az.piva || '';
    document.getElementById('azSede').value = st.az.sede || '';
    document.getElementById('azBadge').textContent = st.az.rag ? `${st.az.rag} (${st.az.piva || 'P.IVA n.d.'})` : 'Azienda cliente da configurare';

    const tc = document.getElementById('tCnt'); tc.innerHTML = '';
    let tot = 0;
    st.cantieri.forEach((c, i) => {
        tot += Number(c.imp || 0);
        const cls = c.stato === 'Attivo' ? 'b-ok' : (c.stato === 'Pre-Cantiere' ? 'b-warn' : 'b-info');
        tc.innerHTML += `<tr><td><strong>${c.cod}</strong></td><td>${c.nome}</td><td>${fmtEur(c.imp)}</td><td>${c.sal}</td><td><span class="badge ${cls}">${c.stato}</span></td><td><button class="btn btn-ghost btn-sm" onclick="delCnt(${i})">X</button></td></tr>`;
    });

    const ta = document.getElementById('tAnag'); ta.innerHTML = '';
    st.anag.forEach((a, i) => {
        ta.innerHTML += `<tr><td><span class="badge b-info">${a.tipo}</span></td><td>${a.nome}</td><td>${a.piva || '-'}</td><td>${a.cont || '-'}</td><td><button class="btn btn-ghost btn-sm" onclick="delAnag(${i})">X</button></td></tr>`;
    });

    document.getElementById('kCnt').textContent = st.cantieri.length;
    document.getElementById('kVal').textContent = fmtEur(tot);
    document.getElementById('kAnag').textContent = st.anag.length;
}

function saveAz() {
    st.az = { rag: document.getElementById('azRag').value.trim(), piva: document.getElementById('azPiva').value.trim(), sede: document.getElementById('azSede').value.trim() };
    persist(); setMsg('mAz', 'Dati azienda salvati.');
}
function addCnt() {
    const cod = document.getElementById('cCod').value.trim() || 'CNT-NEW';
    const nome = document.getElementById('cNome').value.trim();
    const comm = document.getElementById('cComm').value.trim() || '-';
    const imp = parseFloat(document.getElementById('cImp').value || '0');
    const sal = document.getElementById('cSal').value.trim() || '0%';
    const stato = document.getElementById('cStato').value;
    if (!nome) { setMsg('mCnt', 'Inserisci il nome del cantiere.'); return; }
    st.cantieri.push({ cod, nome, comm, imp, sal, stato });
    persist(); setMsg('mCnt', 'Cantiere salvato.');
}
function delCnt(i) { st.cantieri.splice(i, 1); persist(); }
function addAnag() {
    const tipo = document.getElementById('aTipo').value;
    const nome = document.getElementById('aNome').value.trim();
    const piva = document.getElementById('aPiva').value.trim();
    const cont = document.getElementById('aCont').value.trim();
    if (!nome) { setMsg('mAnag', 'Inserisci nome o ragione sociale.'); return; }
    st.anag.push({ tipo, nome, piva, cont });
    persist(); setMsg('mAnag', 'Voce aggiunta alla rubrica.');
}
function delAnag(i) { st.anag.splice(i, 1); persist(); }

EdilControlSuite • Assistenza: 3715132220 | edilcontrolsuite.support@gmail.com`);
}
